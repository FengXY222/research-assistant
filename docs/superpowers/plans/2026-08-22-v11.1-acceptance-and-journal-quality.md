```python
def journal_quality_snapshot(journal: dict) -> dict:
    jcr = journal.get("jcr", {}) if isinstance(journal.get("jcr", {}), dict) else {}
    metrics = jcr.get("metrics", []) if isinstance(jcr.get("metrics", []), list) else []
    primary = metrics[0] if metrics and isinstance(metrics[0], dict) else {}
    quartile = str(primary.get("quartile", "")).strip().upper()
    easy = journal.get("easyscholar", {}) if isinstance(journal.get("easyscholar", {}), dict) else {}
    parts = [f"JCR {quartile}" if quartile else "JCR 待核验"]
    if str(easy.get("cas_upgrade", "")).strip():
        parts.append(f"中科院升级版 {easy['cas_upgrade']}")
    if str(easy.get("impact_factor", "")).strip():
        parts.append(f"IF {easy['impact_factor']}")
    return {"jcr_status": str(jcr.get("status", "pending")), "jcr_quartile": quartile, "cas_upgrade": str(easy.get("cas_upgrade", "")), "cas_basic": str(easy.get("cas_basic", "")), "metric_line": " · ".join(parts), "source": str(jcr.get("source", easy.get("source", ""))), "checked_at": str(easy.get("checked_at", jcr.get("checked_at", ""))), "is_verified": str(jcr.get("status", "")) in {"verified", "manual"}}

def is_default_frontier_eligible(snapshot: dict, *, easyscholar_configured: bool) -> tuple[bool, str]:
    quartile = str(snapshot.get("jcr_quartile", "")).upper()
    if quartile in {"Q1", "Q2"} and bool(snapshot.get("is_verified", False)):
        return True, "verified_q1_q2"
    if quartile in {"Q3", "Q4"}:
        return False, "known_q3_q4"
    return (False, "quality_pending") if easyscholar_configured else (True, "quality_unresolved")
```

Use this module from both `frontier_scoring` and the selection service rather than duplicating quartile/CAS parsing.
